round_half_up <- function(x, digits = 0) {
  posneg <- sign(x)
  z <- abs(x) * 10^digits
  z <- floor(z + 0.5)
  posneg * z / 10^digits
}

find_col <- function(df, target) {
  idx <- which(toupper(names(df)) == toupper(target))
  if (length(idx) == 0) return(NA_character_)
  names(df)[idx[1]]
}

adsl_path <- "D:/meta_hackathon/sas2r-agent/data/raw/cdisc-pilot/adam/adsl.xpt"

if (is.na(eactm_col)) {
  candidates <- c("EACTM", "EACTMN", "ACTM", "ATPTN", "PCTPTNUM", "PCTPT", "ATPT", "AVISITN", "TPTNUM")
  for (cand in candidates) {
    cc <- find_col(adpc, cand)
    if (!is.na(cc)) { eactm_col <- cc; break }
  }
  # last resort: any column containing ACTM
  if (is.na(eactm_col)) {
    hit <- which(grepl("ACTM", toupper(names(adpc))))
    if (length(hit) > 0) eactm_col <- names(adpc)[hit[1]]
  }
}
stopifnot(!is.na(usubjid_col), !is.na(parcat1_col), !is.na(aval_col), !is.na(eactm_col))

    panel.grid.minor = element_blank(),

    csv_df[[cn]] <- as.character(v)
  }
