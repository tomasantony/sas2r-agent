```r
# P040_kmplot.R
# Reimplementation of SAS program P040_kmplot.sas in R
# Produces the same data, Kaplan‑Meier table, and plot.

# ----------------------------------------------------------------------
# Packages ---------------------------------------------------------------
library(survival)
library(ggplot2)
library(dplyr)
library(tidyr)
library(readr)

# ----------------------------------------------------------------------
# Helper: round half up (SAS default) ------------------------------------
round_half_up <- function(x, digits = 0) {
  sign(x) * floor(abs(x) * 10^digits + 0.5) / 10^digits
}

# ----------------------------------------------------------------------
# Output directory -------------------------------------------------------
out_dir <- Sys.getenv("OUT_DIR")
if (!dir.exists(out_dir)) dir.create(out_dir, recursive = TRUE)

# ----------------------------------------------------------------------
# 1. Create the source data set "one" ------------------------------------
one <- bind_rows(
  # Arm 1 uncensored
  tibble(
    arm = "Arm 1",
    censor = 0,
    day = c(143,164,188,188,190,192,206,213,216,220,227,230,
            234,246,265,304)
  ),
  # Arm 1 censored records
  tibble(
    arm = "Arm 1",
    censor = 1,
    day = c(216, 244)
  ),
  # Arm 2 uncensored
  tibble(
    arm = "Arm 2",
    censor = 0,
    day = c(142,156,163,198,205,232,232,233,233,233,233,239,
            240,261,280,280,296,296,323)
  ),
  # Arm 2 censored records
  tibble(
    arm = "Arm 2",
    censor = 1,
    day = c(204, 344)
  )
) %>%
  mutate(
    arm = as.character(arm),
    censor = as.numeric(censor),
    day = as.numeric(day)
  )

# Write the source data set as CSV (mirrors SAS dataset)
write.csv(one,
          file = file.path(out_dir, "one.csv"),
          row.names = FALSE,
          na = "")

# ----------------------------------------------------------------------
# 2. Kaplan‑Meier fit ----------------------------------------------------
fit <- survfit(Surv(day, censor) ~ arm, data = one)

# ----------------------------------------------------------------------
# 3. Full survival table (all event & censor times) --------------------
full_sum <- summary(fit, extend = TRUE)   # no explicit times argument

surv_table <- tibble(
  arm = rep(names(fit$strata), times = fit$strata),
  time = full_sum$time,
  n_risk = full_sum$n.risk,
  n_event = full_sum$n.event,
  n_censor = full_sum$n.censor,
  surv = full_sum$surv,
  std_err = full_sum$std.err,
  lower = full_sum$lower,
  upper = full_sum$upper
) %>%
  arrange(arm, time)

# ----------------------------------------------------------------------
# 4. Write the table as CSV (SAS‑style name) -----------------------------
write.csv(surv_table,
          file = file.path(out_dir, "P040_f_kmplot.csv"),
          row.names = FALSE,
          na = "")

# ----------------------------------------------------------------------
# 5. Plain‑text rendering of the table -----------------------------------
txt_file <- file.path(out_dir, "P040_f_kmplot.txt")
con <- file(txt_file, open = "wt", encoding = "UTF-8")
cat("Kaplan‑Meier Survival Table (generated by R)\n", file = con)

# Header (similar to PROC LIFETEST output)
cat(sprintf("%-6s %6s %8s %8s %8s %12s %12s %12s %12s\n",
            "Arm", "Time", "N_Risk", "N_Event", "N_Censor",
            "Survival", "Std_Err", "Lower_95CI", "Upper_95CI"),
    file = con)

fmt_int <- function(x) formatC(x, format = "d", flag = "#")
fmt_num4 <- function(x) formatC(x, format = "f", digits = 4, flag = "#")

apply(surv_table, 1, function(row) {
  cat(sprintf("%-6s %6s %8s %8s %8s %12s %12s %12s %12s\n",
              row["arm"],
              fmt_int(as.numeric(row["time"])),
              fmt_int(as.numeric(row["n_risk"])),
              fmt_int(as.numeric(row["n_event"])),
              fmt_int(as.numeric(row["n_censor"])),
              fmt_num4(as.numeric(row["surv"])),
              fmt_num4(as.numeric(row["std_err"])),
              fmt_num4(as.numeric(row["lower"])),
              fmt_num4(as.numeric(row["upper"]))),
      file = con)
})
close(con)

# ----------------------------------------------------------------------
# 6. Plot data (step function) -------------------------------------------
plot_df <- surv_table %>%
  select(arm, time, surv) %>%
  mutate(surv = as.numeric(surv))

# Write plotted data as CSV
write.csv(plot_df,
          file = file.path(out_dir, "P040_f_kmplot_plotdata.csv"),
          row.names = FALSE,
          na = "")

# ----------------------------------------------------------------------
# 7. Generate Kaplan‑Meier plot -------------------------------------------
km_plot <- ggplot(plot_df, aes(x = time, y = surv, colour = arm)) +
  geom_step(direction = "hv", size = 1) +
  scale_y_continuous(labels = scales::percent_format(accuracy = 1), limits = c(0, 1)) +
  scale_x_continuous(breaks = seq(0, 330, by = 30), limits = c(0, 330)) +
  labs(
    title = "Example K-M Plot",
    x = "Day",
    y = "Survival Probability",
    colour = "Arm"
  ) +
  theme_minimal() +
  theme(
    plot.title = element_text(hjust = 0.5),
    legend.position = "right"
  )

png_file <- file.path(out_dir, "P040_f_kmplot.png")
ggsave(filename = png_file,
       plot = km_plot,
       width = 8,
       height = 6,
       units = "in",
       dpi = 300)

# ----------------------------------------------------------------------
# End of script -----------------------------------------------------------
```
