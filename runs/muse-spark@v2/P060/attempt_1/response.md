# SAS round half away from zero helper (for fidelity where needed)
round_half_up <- function(x, digits = 0) {
  posneg <- sign(x)
  z <- abs(x) * 10^digits
  z <- floor(z + 0.5 + 1e-10)
  posneg * z / 10^digits
}

find_col <- function(df, target) {
  idx <- which(toupper(names(df)) == toupper(target))
  if (length(idx) == 0) stop(paste0("Column not found: ", target, " in ", paste(names(df), collapse=",")))
  names(df)[idx[1]]
}

  # keep numeric where possible
